`scripts/jogos/tempo.mjs`:

```js
// Horário de Brasília: UTC-3 fixo (o Brasil não tem horário de verão desde 2019).
// Todas as funções trabalham com epoch em SEGUNDOS e datas "AAAA-MM-DD".
export const BRT_OFFSET_SEC = -3 * 3600;

// Dia (AAAA-MM-DD) em Brasília de um instante
export function diaBrt(epochSec) {
  return new Date((epochSec + BRT_OFFSET_SEC) * 1000).toISOString().slice(0, 10);
}

// Epoch da meia-noite de Brasília de um dia
export function inicioDoDiaBrt(diaStr) {
  return Date.parse(diaStr + "T00:00:00Z") / 1000 - BRT_OFFSET_SEC;
}

// Soma (ou subtrai) dias a uma data AAAA-MM-DD
export function somarDias(diaStr, n) {
  return new Date(Date.parse(diaStr + "T00:00:00Z") + n * 86400000).toISOString().slice(0, 10);
}

// Janela de `qtd` dias começando hoje (em Brasília): lista de dias e [inicio, fim) em epoch
export function janelaDias(nowSec, qtd = 3) {
  const hoje = diaBrt(nowSec);
  const dias = [];
  for (let i = 0; i < qtd; i++) dias.push(somarDias(hoje, i));
  return { dias, inicio: inicioDoDiaBrt(dias[0]), fim: inicioDoDiaBrt(somarDias(hoje, qtd)) };
}

// Data e hora de Brasília ("2026-10-26", "19:30:00" ou "19:30") para epoch
export function epochDeBrt(dataStr, horaStr) {
  const hora = /^\d{2}:\d{2}$/.test(horaStr) ? horaStr + ":00" : horaStr;
  return Date.parse(`${dataStr}T${hora}-03:00`) / 1000;
}

// "2026-10-25" -> "20261025" (formato do parâmetro dates da ESPN)
export function aaaammdd(diaStr) {
  return diaStr.replace(/-/g, "");
}
```

`scripts/jogos/times.mjs`:

```js
// Comparação de nomes de times entre fontes (ESPN em inglês/Espanhol, ge e guia em português).
const PALAVRAS_VAZIAS = new Set([
  "fc", "ec", "sc", "cf", "ac", "afc", "cd", "fk", "sv", "ssc",
  "club", "clube", "de", "do", "da", "dos", "das", "the", "football", "futebol",
]);

// Nome -> forma comparável. `aliases` mapeia nome normalizado -> nome canônico normalizado.
export function normalizarTime(nome, aliases = {}) {
  let t = String(nome == null ? "" : nome)
    .normalize("NFD").replace(/[̀-ͯ]/g, "")
    .toLowerCase()
    .replace(/\./g, "")
    .replace(/&/g, " and ")
    .replace(/[^a-z0-9]+/g, " ")
    .trim();
  if (!t) return "";
  if (Object.prototype.hasOwnProperty.call(aliases, t)) t = aliases[t];
  const resto = t.split(" ").filter((p) => p && !PALAVRAS_VAZIAS.has(p)).join(" ");
  return resto || t;
}

// Mesmo time: nomes iguais depois de normalizar, ou o mais curto contido (palavra por palavra) no mais longo
export function mesmoTime(a, b, aliases = {}) {
  const na = normalizarTime(a, aliases);
  const nb = normalizarTime(b, aliases);
  if (!na || !nb) return false;
  if (na === nb) return true;
  const ta = na.split(" ");
  const tb = nb.split(" ");
  const curto = ta.length <= tb.length ? ta : tb;
  const longo = ta.length <= tb.length ? tb : ta;
  return curto.join("").length >= 4 && curto.every((p) => longo.includes(p));
}

// Mesmo jogo: os dois times (mesmo trocados de lado) e início com diferença de até `toleranciaSec`
export function mesmoJogo(a, b, aliases = {}, toleranciaSec = 45 * 60) {
  if (!a || !b || !a.casa || !a.fora || !b.casa || !b.fora) return false;
  if (Math.abs(a.inicio - b.inicio) > toleranciaSec) return false;
  const direto = mesmoTime(a.casa.nome, b.casa.nome, aliases) && mesmoTime(a.fora.nome, b.fora.nome, aliases);
  const trocado = mesmoTime(a.casa.nome, b.fora.nome, aliases) && mesmoTime(a.fora.nome, b.casa.nome, aliases);
  return direto || trocado;
}
```

```js
import { readFileSync } from "node:fs";

export const ESPORTES = ["futebol", "nfl", "ncaaf", "nba", "nhl", "volei", "futsal", "basquete", "lutas", "outros"];

// Lê regras.json (ao lado deste arquivo, ou de `caminho`) e valida
export function carregarRegras(caminho) {
  const url = caminho || new URL("./regras.json", import.meta.url);
  const regras = JSON.parse(readFileSync(url, "utf8"));
  validarRegras(regras);
  return regras;
}

// Erro claro para regra quebrada: o gerador falha em vez de publicar lixo
export function validarRegras(r) {
  if (!r || typeof r !== "object") throw new Error("regras: arquivo inválido");
  if (!r.fontes || ["espn", "ge", "guia"].some((k) => typeof r.fontes[k] !== "boolean")) {
    throw new Error("regras: fontes precisa de espn, ge e guia booleanos");
  }
  if (!r.duracaoHoras || ESPORTES.some((e) => !(r.duracaoHoras[e] > 0))) {
    throw new Error("regras: duracaoHoras precisa de um número positivo por esporte");
  }
  if (!Array.isArray(r.competicoes) || !r.competicoes.length) throw new Error("regras: competicoes vazia");
  const ids = new Set();
  for (const c of r.competicoes) {
    if (!c.id || ids.has(c.id)) throw new Error(`regras: id de competição repetido ou vazio (${c.id})`);
    ids.add(c.id);
    if (!ESPORTES.includes(c.esporte)) throw new Error(`regras: ${c.id} com esporte inválido (${c.esporte})`);
    if (!/^[a-z-]+\/[a-z0-9._-]+$/.test(String(c.espn))) throw new Error(`regras: ${c.id} com caminho da ESPN inválido`);
    if (!c.nome || !Array.isArray(c.provaveis) || !c.provaveis.length) throw new Error(`regras: ${c.id} sem nome ou sem canais prováveis`);
  }
  if (!r.marcas || typeof r.marcas !== "object" || Object.values(r.marcas).some((v) => !Array.isArray(v))) {
    throw new Error("regras: marcas precisa mapear nome para lista de canais");
  }
  if (!Array.isArray(r.esportePorPalavra)) throw new Error("regras: esportePorPalavra precisa ser lista");
  for (const p of r.esportePorPalavra) {
    new RegExp(p.padrao, "i"); // lança se não compilar
    if (!ESPORTES.includes(p.esporte)) throw new Error(`regras: esportePorPalavra com esporte inválido (${p.esporte})`);
  }
}
```

```js
import { aaaammdd } from "./tempo.mjs";

const BASE = "https://site.api.espn.com/apis/site/v2/sports/";

// Endereço do placar de uma competição em um dia (a ESPN aceita UMA data por consulta, no formato AAAAMMDD)
export function urlEspn(competicao, diaStr) {
  const extra = competicao.consulta ? "&" + competicao.consulta : "";
  return `${BASE}${competicao.espn}/scoreboard?dates=${aaaammdd(diaStr)}${extra}`;
}

// Resposta da ESPN -> lista de jogos no formato interno (canais vazios; prováveis = os da competição)
export function parseEspn(json, competicao, duracaoHoras) {
  const eventos = json && Array.isArray(json.events) ? json.events : [];
  const durSec = (duracaoHoras[competicao.esporte] || duracaoHoras.outros || 2.5) * 3600;
  const jogos = [];
  for (const e of eventos) {
    const comp = e && Array.isArray(e.competitions) ? e.competitions[0] : null;
    const lados = comp && Array.isArray(comp.competitors) ? comp.competitors : [];
    const inicio = Math.floor(Date.parse(e && e.date) / 1000);
    if (!e || !e.id || !isFinite(inicio) || lados.length < 2) continue;
    const casaRaw = lados.find((l) => l.homeAway === "home") || lados[0];
    const foraRaw = lados.find((l) => l.homeAway === "away") || lados[1];
    const lado = (l) => ({
      nome: String((l.team && (l.team.displayName || l.team.shortDisplayName)) || ""),
      escudo: String((l.team && l.team.logo) || ""),
    });
    const casa = lado(casaRaw);
    const fora = lado(foraRaw);
    if (!casa.nome || !fora.nome) continue;

// Consulta todas as competições em todos os dias (4 em paralelo). Falha de uma consulta não derruba as outras.
// fetchFn(url) devolve algo como a Response do fetch ({ ok, status, json() }).
export async function buscarEspn({ competicoes, dias, fetchFn, duracaoHoras, avisar }) {
  const tarefas = [];
  competicoes.forEach((c) => dias.forEach((d) => tarefas.push({ c, d })));
  const porId = new Map();
  let ok = 0;
  let falhas = 0;
  let proximo = 0;
  async function trabalhador() {
    while (proximo < tarefas.length) {
      const t = tarefas[proximo++];
      try {
        const r = await fetchFn(urlEspn(t.c, t.d));
        if (!r.ok) throw new Error("HTTP " + r.status);
        const json = await r.json();
        parseEspn(json, t.c, duracaoHoras).forEach((j) => { if (!porId.has(j.idFonte)) porId.set(j.idFonte, j); });
        ok++;
      } catch (e) {
        falhas++;
        avisar(`ESPN ${t.c.id} ${t.d}: ${e.message}`);
      }
    }
  }
  await Promise.all(Array.from({ length: 4 }, trabalhador));
  return { jogos: [...porId.values()], ok, falhas };
}
```

```js
// Jogos a partir do guia de TV (epg-extra.json): { canais: { nome: [[inicio, fim, titulo, descricao], ...] } }
const REPRISE = /^\s*(vt|reprise|compacto|melhores momentos|p[oó]s[- ]jogo|pr[eé][- ]jogo)\b/i;
const CONFRONTO = /^(.*?)\s+(?:x|vs\.?|versus)\s+(.*)$/i;

// Esporte e nome da competição a partir do prefixo do título ("NFL", "Liga Evolución de Futsal"...)
function classificar(prefixo, regras) {
  if (!prefixo) return { esporte: "futebol", competicao: "Futebol" };
  for (const r of regras.esportePorPalavra) {
    if (new RegExp(r.padrao, "i").test(prefixo)) return { esporte: r.esporte, competicao: r.competicao || prefixo };
  }
  return { esporte: "outros", competicao: prefixo };
}

// Título de programa -> { esporte, competicao, casa, fora } ou null se não for um jogo ao vivo
export function interpretarTitulo(titulo, regras) {
  let t = String(titulo == null ? "" : titulo).trim();
  if (!t || REPRISE.test(t)) return null;
  t = t.replace(/\s*[-–—]\s*ao vivo\s*$/i, "").trim();
  let prefixo = "";
  const m = t.match(/^([^:]{2,60}):\s+(.*)$/);
  if (m) { prefixo = m[1].trim(); t = m[2].trim(); }
  const c = t.match(CONFRONTO);
  if (!c) return null;
  const casa = c[1].trim();
  const fora = c[2].trim();
  if (!casa || !fora) return null;
  const { esporte, competicao } = classificar(prefixo, regras);
  return { esporte, competicao, casa, fora };
}

// Todos os jogos do guia, um por programa (o mesmo jogo em dois canais vira dois itens; a mescla junta depois)
export function jogosDoGuia(epg, regras) {
  const canais = epg && epg.canais && typeof epg.canais === "object" ? epg.canais : {};
  const saida = [];
  for (const canal of Object.keys(canais)) {
    const programas = Array.isArray(canais[canal]) ? canais[canal] : [];
    for (const p of programas) {
      if (!Array.isArray(p)) continue;
      const j = interpretarTitulo(p[2], regras);
      if (!j || !isFinite(p[0]) || !isFinite(p[1])) continue;
      saida.push({
        fonte: "guia",
        idFonte: "",
        esporte: j.esporte,
        competicao: j.competicao,

        fora: { nome: j.fora, escudo: "" },
        inicio: p[0],
```

```js
import { epochDeBrt } from "./tempo.mjs";

// Endereço da agenda do ge (a página embute os dados em window.dataSportsSchedule)
export const URL_AGENDA_GE = "https://ge.globo.com/agenda/";

// Lê um objeto JSON a partir da "{" na posição `ini`, respeitando strings e barras de escape
function lerObjetoJson(s, ini) {
  let profundidade = 0;
  let emTexto = false;
  let escape = false;
  for (let p = ini; p < s.length; p++) {
    const ch = s[p];
    if (emTexto) {
      if (escape) escape = false;
      else if (ch === "\\") escape = true;
      else if (ch === '"') emTexto = false;
      continue;
    }
    if (ch === '"') emTexto = true;
    else if (ch === "{") profundidade++;
    else if (ch === "}") {
      profundidade--;
      if (profundidade === 0) return s.slice(ini, p + 1);
    }
  }
  throw new Error("agenda do ge: JSON incompleto");
}

// HTML da página -> objeto { "AAAA-MM-DD": { championshipsAgenda: [...] } }
export function extrairAgendaGe(html) {
  const i = html.indexOf("window.dataSportsSchedule");
  if (i < 0) throw new Error("agenda do ge: bloco window.dataSportsSchedule não encontrado");
  const k = html.indexOf("sport:", i);
  if (k < 0) throw new Error("agenda do ge: campo sport não encontrado");
  const abre = html.indexOf("{", k);
  if (abre < 0) throw new Error("agenda do ge: campo sport sem objeto");
  return JSON.parse(lerObjetoJson(html, abre));
}

// Agenda -> jogos de futebol no formato interno. As marcas (SporTV, Premiere...) viram canais prováveis pela tabela `marcas`.
export function jogosDoGe(agenda, regras) {
  const saida = [];
  const vistos = new Set();
  const dur = (regras.duracaoHoras.futebol || 2) * 3600;
  for (const dia of Object.keys(agenda || {})) {
    const agendaDia = agenda[dia];
    const campeonatos = agendaDia && Array.isArray(agendaDia.championshipsAgenda) ? agendaDia.championshipsAgenda : [];
    for (const camp of campeonatos) {
      const nomeCamp = String((camp.championship && camp.championship.name) || "");
      for (const balde of ["past", "now", "future"]) {
        for (const ev of Array.isArray(camp[balde]) ? camp[balde] : []) {
          const m = ev && ev.match;
          if (!m || !m.firstContestant || !m.secondContestant) continue;
          const inicio = epochDeBrt(m.startDate || dia, m.startHour || "00:00:00");
          const casaNome = m.firstContestant.popularName || m.firstContestant.name;
          const foraNome = m.secondContestant.popularName || m.secondContestant.name;
          if (!isFinite(inicio) || !casaNome || !foraNome) continue;
          const id = String(m.id);
          if (vistos.has(id)) continue;
          vistos.add(id);
          const provaveis = [];
          for (const fonte of Array.isArray(m.liveWatchSources) ? m.liveWatchSources : []) {
            const canais = regras.marcas[String(fonte.name || "").toLowerCase()] || [];
            canais.forEach((c) => { if (!provaveis.includes(c)) provaveis.push(c); });
          }
          saida.push({
            fonte: "ge",
            idFonte: id,
            esporte: "futebol",
            competicao: nomeCamp,
            casa: { nome: String(casaNome), escudo: String(m.firstContestant.badgePng || "") },
            fora: { nome: String(foraNome), escudo: String(m.secondContestant.badgePng || "") },
            inicio,
            fim: inicio + dur,
            canais: [],
            provaveis,
          });
        }
      }
    }
  }
  return saida;
}
```

```js
import { mesmoJogo, normalizarTime } from "./times.mjs";

const TOLERANCIA_SEC = 45 * 60;

// Hash curto e estável (FNV-1a de 32 bits) para o id dos jogos que só o guia conhece
function hash(texto) {
  let h = 0x811c9dc5;
  for (let i = 0; i < texto.length; i++) {
    h ^= texto.charCodeAt(i);
    h = Math.imul(h, 0x01000193) >>> 0;
  }
  return h.toString(16).padStart(8, "0");
}

// Id do jogo: ESPN e ge usam o id da fonte; o guia usa os times (em ordem alfabética, então a ordem casa/fora não importa) e o início
export function idDoJogo(j, aliases) {
  if (j.fonte === "espn") return `espn-${j.idFonte}`;
  if (j.fonte === "ge") return `ge-${j.idFonte}`;
  const times = [normalizarTime(j.casa.nome, aliases), normalizarTime(j.fora.nome, aliases)].sort();
  return `guia-${hash(`${times[0]}|${times[1]}|${j.inicio}`)}`;
}

// Junta as fontes. A ESPN é a base; o ge acrescenta as marcas de transmissão (e os jogos que a ESPN não tem);
// o guia confirma o canal (e traz os jogos que só ele conhece). Depois filtra pela janela e ordena.
export function mesclarJogos({ espn, ge, guia, aliases, janela }) {
  const base = espn.map((j) => ({ ...j, canais: j.canais.slice(), provaveis: j.provaveis.slice() }));
  const acha = (j) => base.find((x) => mesmoJogo(x, j, aliases, TOLERANCIA_SEC));

  for (const g of ge) {
    const alvo = acha(g);
    if (alvo) {
      if (g.provaveis.length) alvo.provaveis = g.provaveis.slice();
    } else {
      base.push({ ...g, canais: [], provaveis: g.provaveis.slice() });
    }
  }
  for (const gu of guia) {
    const alvo = acha(gu);
    if (alvo) {
      gu.canais.forEach((c) => { if (!alvo.canais.includes(c)) alvo.canais.push(c); });
    } else {
      base.push({ ...gu, canais: gu.canais.slice(), provaveis: [] });
    }
  }

  return base
    .filter((j) => j.inicio >= janela.inicio && j.inicio < janela.fim)
    .map((j) => ({
      id: idDoJogo(j, aliases),
      esporte: j.esporte,
      competicao: j.competicao,
      casa: j.casa,
      fora: j.fora,
      inicio: j.inicio,
      fim: j.fim,
      canais: j.canais,
      provaveis: j.canais.length ? [] : j.provaveis,
    }))
    .sort((a, b) => a.inicio - b.inicio || a.id.localeCompare(b.id));
}
```

```js
const JOGOS_URL = "https://raw.githubusercontent.com/Guaitolinii/visual-craft-assistant/vod-data/jogos.json";
const JOGOS_CACHE_KEY = "sint_jogos_cache";
const JOGOS_CACHE_MAX_MS = 24 * 3600 * 1000;   // cópia guardada vale por 24 h se a rede falhar
const JOGOS_REFRESH_MS = 20 * 60 * 1000;       // baixa de novo se a cópia tiver mais de 20 min
const JOGOS_SOBRA_MS = 10 * 60 * 1000;         // o jogo some 10 min depois do fim
const JOGOS_ROTULOS = { futebol: "Futebol", nfl: "NFL", ncaaf: "College Football", nba: "NBA", nhl: "NHL", volei: "Vôlei", futsal: "Futsal", basquete: "Basquete", lutas: "Lutas", outros: "Outros" };
const JOGOS_ORDEM_OUTROS = ["nfl", "ncaaf", "nba", "nhl", "volei", "futsal", "basquete", "lutas", "outros"];

// Dia (AAAA-MM-DD) e hora (HH:MM) de Brasília, UTC-3 fixo: o app mostra sempre o horário de Brasília, em qualquer aparelho
function jogosDiaBrt(epochSec) {
  return new Date((epochSec - 3 * 3600) * 1000).toISOString().slice(0, 10);
}
function jogosHoraBrt(epochSec) {
  return new Date((epochSec - 3 * 3600) * 1000).toISOString().slice(11, 16);
}

function jogosAoVivo(j, nowMs) {
  const t = nowMs / 1000;
  return j.inicio <= t && t < j.fim;
}
function jogosVisivel(j, nowMs) {
  return nowMs / 1000 < j.fim + JOGOS_SOBRA_MS / 1000;
}

// Seções da tela: futebol em "ao vivo agora", "hoje" e "amanhã"; os outros esportes em grupos (NFL, College Football, NBA, NHL e demais).
// Jogo que começou ontem e ainda está no ar entra em "ao vivo". Jogo terminado há mais de 10 min e jogo de depois de amanhã não entram.
function jogosSecoes(jogos, nowMs) {
  const vazio = { aoVivo: [], hoje: [], amanha: [], outros: [] };
  if (!Array.isArray(jogos)) return vazio;
  const hoje = jogosDiaBrt(nowMs / 1000);
  const amanha = jogosDiaBrt(nowMs / 1000 + 86400);
  const lista = jogos
    .filter((j) => j && jogosVisivel(j, nowMs) && (jogosAoVivo(j, nowMs) || [hoje, amanha].includes(jogosDiaBrt(j.inicio))))
    .sort((a, b) => a.inicio - b.inicio || String(a.id).localeCompare(String(b.id)));
  const futebol = lista.filter((j) => j.esporte === "futebol");
  const resto = futebol.filter((j) => !jogosAoVivo(j, nowMs));
  return {
    aoVivo: futebol.filter((j) => jogosAoVivo(j, nowMs)),
    hoje: resto.filter((j) => jogosDiaBrt(j.inicio) === hoje),
    amanha: resto.filter((j) => jogosDiaBrt(j.inicio) === amanha),
    outros: JOGOS_ORDEM_OUTROS.map((esporte) => ({
      esporte,
      rotulo: JOGOS_ROTULOS[esporte],
      itens: lista.filter((j) => j.esporte === esporte).map((j) => ({
        jogo: j,
        dia: jogosDiaBrt(j.inicio) === hoje ? "hoje" : "amanha",
        aoVivo: jogosAoVivo(j, nowMs),
      })),
    })).filter((g) => g.itens.length),
  };
}

// Qualidade da versão de um canal: HD primeiro, depois FHD, depois SD, depois o resto
function jogosPesoVariante(nome) {
  const n = String(nome == null ? "" : nome).toLowerCase();
  if (/\bhd\b/.test(n)) return 0;
  if (/\bfhd\b/.test(n)) return 1;
  if (/\bsd\b/.test(n)) return 2;
  return 3;
}

// Canais da conta que casam com os nomes do jogo. As versões HD/FHD/SD do mesmo canal contam como um só (vale a melhor).
function jogosCanaisDaConta(jogo, canais) {
  const porNome = new Map();
  (Array.isArray(canais) ? canais : []).forEach((ch) => {
    const n = normalizeChannelName(ch && ch.name);
    if (!n) return;
    const atual = porNome.get(n);
    if (!atual || jogosPesoVariante(ch.name) < jogosPesoVariante(atual.name)) porNome.set(n, ch);
  });
  const pega = (nomes) => {
    const saida = [];
    const vistos = new Set();
    (Array.isArray(nomes) ? nomes : []).forEach((nome) => {
      const n = normalizeChannelName(nome);
      const ch = porNome.get(n);
      if (ch && !vistos.has(n)) { vistos.add(n); saida.push(ch); }
    });
    return saida;
  };
  return { confirmados: pega(jogo.canais), provaveis: pega(jogo.provaveis) };
}

// O que fazer ao tocar no card: "tocar" (um canal confirmado na lista), "escolher" (vários, ou só prováveis) ou "nenhum"
function jogosAcaoClique(jogo, canais) {
  const r = jogosCanaisDaConta(jogo, canais);
  if (r.confirmados.length === 1) return { tipo: "tocar", canal: r.confirmados[0] };
  const opcoes = r.confirmados.length ? r.confirmados : r.provaveis;
  if (!opcoes.length) return { tipo: "nenhum" };
  return { tipo: "escolher", canais: opcoes };
}

// Confere o formato do arquivo (v1) e descarta jogo quebrado. null = arquivo inválido.
function jogosValidar(dados) {
  if (!dados || dados.v !== 1 || !Array.isArray(dados.jogos)) return null;
  const jogos = dados.jogos.filter((j) => j && typeof j.id === "string" && typeof j.esporte === "string"
    && isFinite(j.inicio) && isFinite(j.fim) && Array.isArray(j.canais) && Array.isArray(j.provaveis));
  return { v: 1, gerado: dados.gerado, jogos };
}

// Cópia guardada no aparelho: { ts, data } com menos de 24 h e íntegra, senão null
function jogosLerCache(raw, nowMs) {
  let o = null;
  try { o = JSON.parse(raw || "null"); } catch (e) { return null; }
  if (!o || !isFinite(o.ts) || nowMs - o.ts > JOGOS_CACHE_MAX_MS) return null;
  const data = jogosValidar(o.data);
  return data ? { ts: o.ts, data } : null;
}
```

```js
async function jogosBaixar() {

async function loadJogos(force) {
```

```ts
// Mesmas regras do app web (sintoniza-link.html). Sintaxe até ES2016 (navegador da TV, Chrome 53).
import { normalizeChannelName } from './epg';
import { fetchText, JOGOS_URL } from './api';
import type { Channel } from '../types/tv';

export interface JogoTime { nome: string; escudo: string }
export interface Jogo {
  id: string;
  esporte: string;
  competicao: string;
  titulo?: string;
  casa?: JogoTime;
  fora?: JogoTime;
  inicio: number; // epoch em segundos
  fim: number;
  canais: string[];
  provaveis: string[];
}
export interface ArquivoJogos { v: number; gerado: number; jogos: Jogo[] }
export interface JogoItem { jogo: Jogo; dia: 'hoje' | 'amanha'; aoVivo: boolean }
export interface GrupoOutros { esporte: string; rotulo: string; itens: JogoItem[] }
export interface SecoesJogos { aoVivo: Jogo[]; hoje: Jogo[]; amanha: Jogo[]; outros: GrupoOutros[] }
export type AcaoClique =
  | { tipo: 'tocar'; canal: Channel }
  | { tipo: 'escolher'; canais: Channel[] }
  | { tipo: 'nenhum' };

const CACHE_KEY = 'sint_jogos_cache';
const CACHE_MAX_MS = 24 * 3600 * 1000;  // a cópia guardada vale por 24 h se a rede falhar
const REFRESH_MS = 20 * 60 * 1000;      // baixa de novo se a cópia tiver mais de 20 min
const SOBRA_MS = 10 * 60 * 1000;        // o jogo some 10 min depois do fim
const TIMEOUT_MS = 10000;

const ROTULOS: Record<string, string> = {
  futebol: 'Futebol', nfl: 'NFL', ncaaf: 'College Football', nba: 'NBA', nhl: 'NHL',
  volei: 'Vôlei', futsal: 'Futsal', basquete: 'Basquete', lutas: 'Lutas', outros: 'Outros',
};
const ORDEM_OUTROS = ['nfl', 'ncaaf', 'nba', 'nhl', 'volei', 'futsal', 'basquete', 'lutas', 'outros'];

// Dia (AAAA-MM-DD) e hora (HH:MM) de Brasília, UTC-3 fixo: a TV mostra sempre o horário de Brasília
export function diaBrt(epochSec: number): string {
  return new Date((epochSec - 3 * 3600) * 1000).toISOString().slice(0, 10);
}
export function horaBrt(epochSec: number): string {
  return new Date((epochSec - 3 * 3600) * 1000).toISOString().slice(11, 16);
}

export function aoVivo(j: Jogo, nowMs: number): boolean {
  const t = nowMs / 1000;
  return j.inicio <= t && t < j.fim;
}
export function visivel(j: Jogo, nowMs: number): boolean {
  return nowMs / 1000 < j.fim + SOBRA_MS / 1000;
}

export function tituloDoJogo(j: Jogo): string {
  return j.casa && j.fora ? j.casa.nome + ' x ' + j.fora.nome : (j.titulo || j.competicao);
}

// Seções da tela: futebol em "ao vivo agora", "hoje" e "amanhã"; os outros esportes em grupos.
// Jogo que começou ontem e ainda está no ar entra em "ao vivo". Terminado há mais de 10 min e depois de amanhã não entram.
export function secoes(jogos: Jogo[] | null | undefined, nowMs: number): SecoesJogos {
  if (!Array.isArray(jogos)) return { aoVivo: [], hoje: [], amanha: [], outros: [] };
  const hoje = diaBrt(nowMs / 1000);
  const amanha = diaBrt(nowMs / 1000 + 86400);
  const lista = jogos
    .filter((j) => j && visivel(j, nowMs) && (aoVivo(j, nowMs) || hoje === diaBrt(j.inicio) || amanha === diaBrt(j.inicio)))
    .sort((a, b) => a.inicio - b.inicio || String(a.id).localeCompare(String(b.id)));
  const futebol = lista.filter((j) => j.esporte === 'futebol');
  const resto = futebol.filter((j) => !aoVivo(j, nowMs));
  const outros: GrupoOutros[] = [];
  ORDEM_OUTROS.forEach((esporte) => {
    const itens: JogoItem[] = lista
      .filter((j) => j.esporte === esporte)
      .map((j) => ({ jogo: j, dia: (diaBrt(j.inicio) === hoje ? 'hoje' : 'amanha') as 'hoje' | 'amanha', aoVivo: aoVivo(j, nowMs) }));
    if (itens.length) outros.push({ esporte, rotulo: ROTULOS[esporte], itens });
  });
  return {
    aoVivo: futebol.filter((j) => aoVivo(j, nowMs)),
    hoje: resto.filter((j) => diaBrt(j.inicio) === hoje),
    amanha: resto.filter((j) => diaBrt(j.inicio) === amanha),
    outros,
  };
}

// Qualidade da versão de um canal: HD primeiro, depois FHD, depois SD, depois o resto
export function pesoVariante(nome: string | null | undefined): number {
  const n = String(nome == null ? '' : nome).toLowerCase();
  if (/\bhd\b/.test(n)) return 0;
  if (/\bfhd\b/.test(n)) return 1;
  if (/\bsd\b/.test(n)) return 2;
  return 3;
}

// Canais da conta que casam com os nomes do jogo. As versões HD/FHD/SD do mesmo canal contam como um só (vale a melhor).
export function canaisDaConta(jogo: Jogo, canais: Channel[] | null | undefined): { confirmados: Channel[]; provaveis: Channel[] } {
  const porNome: Record<string, Channel> = {};
  (Array.isArray(canais) ? canais : []).forEach((ch) => {
    const n = normalizeChannelName(ch && ch.name);
    if (!n) return;
    const atual = porNome[n];
    if (!atual || pesoVariante(ch.name) < pesoVariante(atual.name)) porNome[n] = ch;
  });
  const pega = (nomes: string[]): Channel[] => {
    const saida: Channel[] = [];
    const vistos: Record<string, boolean> = {};
    (Array.isArray(nomes) ? nomes : []).forEach((nome) => {
      const n = normalizeChannelName(nome);
      const ch = porNome[n];
      if (ch && !vistos[n]) { vistos[n] = true; saida.push(ch); }
    });
    return saida;
  };
  return { confirmados: pega(jogo.canais), provaveis: pega(jogo.provaveis) };
}

// O que fazer ao selecionar o card: tocar (um canal confirmado na lista), escolher (vários, ou só prováveis) ou nenhum
export function acaoClique(jogo: Jogo, canais: Channel[] | null | undefined): AcaoClique {
  const r = canaisDaConta(jogo, canais);
  if (r.confirmados.length === 1) return { tipo: 'tocar', canal: r.confirmados[0] };
  const opcoes = r.confirmados.length ? r.confirmados : r.provaveis;
  if (!opcoes.length) return { tipo: 'nenhum' };
  return { tipo: 'escolher', canais: opcoes };
}

// Confere o formato do arquivo (v1) e descarta jogo quebrado. null = arquivo inválido.
export function validar(dados: any): ArquivoJogos | null {
  if (!dados || dados.v !== 1 || !Array.isArray(dados.jogos)) return null;
  const jogos = (dados.jogos as any[]).filter(
    (j) => j && typeof j.id === 'string' && typeof j.esporte === 'string' && isFinite(j.inicio) && isFinite(j.fim) && Array.isArray(j.canais) && Array.isArray(j.provaveis)
  );
  return { v: 1, gerado: dados.gerado, jogos: jogos as Jogo[] };
}

// Cópia guardada no aparelho: { ts, dados } com menos de 24 h e íntegra, senão null
export function lerCache(raw: string | null | undefined, nowMs: number): { ts: number; dados: ArquivoJogos } | null {
  let o: any = null;
  try { o = JSON.parse(raw || 'null'); } catch (e) { return null; }
  if (!o || !isFinite(o.ts) || nowMs - o.ts > CACHE_MAX_MS) return null;
  const dados = validar(o.data);
  return dados ? { ts: o.ts, dados } : null;
}

let memoria: { ts: number; dados: ArquivoJogos } | null = null;

// Carrega os jogos: usa a cópia (memória ou aparelho) se tiver menos de 20 min; senão baixa. Falha de rede devolve a cópia que houver (ou null).
export function carregarJogos(forcar: boolean): Promise<ArquivoJogos | null> {
  const agora = Date.now();
  if (!memoria) {
    let raw: string | null = null;
    try { raw = localStorage.getItem(CACHE_KEY); } catch (e) { raw = null; }
    const c = lerCache(raw, agora);
    if (c) memoria = c;
  }
  if (!forcar && memoria && agora - memoria.ts <= REFRESH_MS) return Promise.resolve(memoria.dados);
  return fetchText(JOGOS_URL, TIMEOUT_MS).then(
    (texto): ArquivoJogos | null => {
      let json: unknown = null;
      try { json = JSON.parse(texto.replace(/^\uFEFF/, '')); } catch (e) { json = null; }
      const dados = validar(json);
      if (!dados) return memoria ? memoria.dados : null;
      memoria = { ts: Date.now(), dados };
      try { localStorage.setItem(CACHE_KEY, JSON.stringify({ ts: memoria.ts, data: dados })); } catch (e) { /* sem armazenamento */ }
      return dados;
    },
    (): ArquivoJogos | null => (memoria ? memoria.dados : null)
  );
}
```
